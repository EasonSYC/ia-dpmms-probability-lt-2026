\section{Multivariate Continuous Random Variables}

\subsection{Distribution, Density and Independence}

\begin{definition}[Distribution and Density of Joint Continuous Random Variable]
    \label{def:distribution-density-joint-crv}%
    The joint continuous random variable \(\vm{X} = \para{X_1, \ldots, X_n} \in \RR^n\) has \emph{probability density function} \(f\), if for all \(x_1, \ldots, x_n\), that
    \[
        \Prob\para{X_1 \leq x_1, \ldots, X_n \leq x_n} = \int_{\minusinfty}^{x_1} \Diff y_1 \cdots \int_{\minusinfty}^{x_n} \Diff y_n f\para{y_1, \ldots, y_n}.
    \]

    The left-hand side
    \[
        F \para{x_1, \ldots, x_n} = \Prob\para{X_1 \leq x_1, \ldots, X_n \leq x_n}
    \]
    is the \emph{probability distribution function} of \(\vm{X}\).
\end{definition}

We have
\[
    f\para{x_1, \ldots, x_n} = \frac{\partial^n F}{\partial x_1 \cdots \partial x_n} \para{x_1, \ldots, x_n}.
\]

For all \(\mathcal{B} \subseteq \RR^n\), we have
\[
    \Prob\para{\para{X_1, \ldots, X_n} \in \mathcal{B}} = \int_{\mathcal{B}} f \para{y_1, \ldots, y_n} \Diff^n \vm{y}.
\]

\begin{definition}[Independence of Continuous Random Vaiables]
    \label{def:independence-continuous-random-variables}%
    We say the continuous random variables \(X_1, \ldots, X_n\) are \emph{independent}, if for all \(x_1, \ldots, x_n\), we have
    \[
        \Prob\para{X_1 \leq x_1, \ldots, X_n \leq x_n} = \Prob\para{X_1 \leq x_1} \cdots \Prob\para{X_n \leq x_n}.
    \]
\end{definition}

\begin{theorem}[Independence if and only if Density is Seperable]
    \label{thm:rv-independent-density-separable}%
    Let \(\vm{X} = \para{X_1, \ldots, X_n}\) have density \(f\).
    \begin{enumerate}
        \item Suppose \(X_1, \ldots, X_n\) are independent, with densities \(f_1, \ldots, f_n\) respectively. Then
        \[
            f\para{x_1, \ldots, x_n} = f_1\para{x_1} \cdots f_n\para{x_n}
        \]
        for all \(x_1, \ldots, x_n\).
        \item Suppose \(f\) factorises into above, for some non-negative functions \(f_1, \ldots, f_n\). Then \(X_1, \ldots, X_n\) are independent, with densities proportional to \(f_1, \ldots, f_n\) respectively (i.e. up to normalisation).
    \end{enumerate}
\end{theorem}

\begin{proof}[of First Part]
    We investigate the distribution function. We have
    \begin{align*}
        \Prob\para{X_1 \leq x_1, \ldots, X_n \leq x_n} &= \Prob\para{X_1 \leq x_1} \cdots \Prob\para{X_n \leq x_n} \\
        &= \int_{\minusinfty}^{x_1} f_1 \para{y_1} \Diff y_1 \cdots \int_{\minusinfty}^{x_n} f_n \para{y_n} \Diff y_n \\
        &= \int_{\minusinfty}^{x_1} \Diff y_1 \cdots \int_{\minusinfty}^{x_n} \Diff y_n f_1\para{y_1} f_2\para{y_2} \cdots f_n\para{y_n},
    \end{align*}
    so
    \[
        f\para{x_1, \ldots, x_n} = f_1\para{x_1} \cdots f_n\para{x_n}.
    \]
\end{proof}

\begin{proof}[of Second Part]
    We have that
    \begin{align*}
        \Prob\para{X_1 \leq x_1, \ldots, X_n \leq x_n} &= \int_{\minusinfty}^{x_1} \Diff y_1 \cdots \int_{\minusinfty}^{x_n} \Diff y_n f\para{y_1, \ldots, y_n} \\
        &= \int_{\minusinfty}^{x_1} \Diff y_1 \cdots \int_{\minusinfty}^{x_n} \Diff y_n f_1\para{y_1} \cdots f_n\para{y_n} \\
        &= \int_{\minusinfty}^{x_1} f_1\para{y_1} \Diff y_1 \cdots \int_{\minusinfty}^{x_n} f_n\para{y_n} \Diff y_n.
    \end{align*}

    But we have that
    \[
        \int_{\minusinfty}^{\plusinfty} \Diff x_1 \cdots \int_{\minusinfty}^{\plusinfty} \Diff x_n f_1\para{x_1} \cdots f_n\para{x_2} = 1,
    \]
    and hence
    \[
        \Prob\para{X_1 \leq x_1, \ldots, X_n \leq x_n} = \frac{\int_{\minusinfty}^{x_1} f_1\para{y_1} \Diff y_1}{\int_{\minusinfty}^{\plusinfty} f_1\para{y_1} \Diff y_1} \cdots \frac{\int_{\minusinfty}^{x_1} f_n\para{y_n} \Diff y_n}{\int_{\minusinfty}^{\plusinfty} f_n\para{y_n} \Diff y_n}.
    \]

    So \(X_1, \ldots, X_n\) are independent, with densities
    \[
        \frac{f_i}{\int_{\minusinfty}^{\plusinfty} f_i\para{x} \Diff x}
    \]
    respectively.
\end{proof}

\subsection{Marginal Distribution and Density}

\begin{definition}[Marginal Distribution and Density]
    \label{def:marginal-distribution-density}%
    For a joint continuous random variable \(\vm{X} = \para{X_1, X_2, \ldots, X_n}\), the \emph{marginal distribution} \(F_{X_i}\) and the \emph{marginal density} \(f_{X_i}\) of \(X_i\) are given by, respectively,
    \[
        F_{X_i} \para{x} = \Prob\para{X_i \leq x} = \Prob\para{X_1 \in \RR, \ldots, X_{i - 1} \in \RR, X_i \leq x, X_{i + 1} \in \RR, \ldots, X_n \in \RR}
    \]
    and
    \[
        f_{X_i} \para{x} = F_{X_i}'\para{x}.
    \]
\end{definition}

\begin{proposition}[Integration Formula for Marginal Density and Distribution]
    \label{prop:marginal-distribution-density-formula}%
    Suppose \(\vm{X} = \para{X_1, \ldots, X_n}\) has density \(f\). Then we have
    \[
        F_{X_1} \para{x} = \int_{\minusinfty}^{x} \Diff x_1 \cdot \int_{\minusinfty}^{\plusinfty} \Diff x_2 \cdots \int_{\minusinfty}^{\plusinfty} \Diff x_n f\para{x_1, \ldots, x_n},
    \]
    and
    \[
        f_{X_1} \para{x} = \int_{\minusinfty}^{\plusinfty} \Diff x_2 \cdots \int_{\minusinfty}^{\plusinfty} \Diff x_n f\para{x, \ldots, x_n}.
    \]
\end{proposition}

\subsection{Sum of Independent Random Variables}

Suppose \(X\) and \(Y\) are independent random variables with densities \(f_X\) and \(f_Y\) respectively. What is the density of \(X + Y\)?

In the discrete case, we recall this is given by the convolution
\[
    \Prob\para{X + Y = z} = \sum_{x = \minusinfty}^{\plusinfty} \Prob\para{X = x} \cdot \Prob\para{Y = z - x}.
\]

The continuous case is similar.

\begin{proposition}[Distribution of Sum of Independent Continuous Random Variables]
    \label{prop:dist-sum-indp-crv}%
    Let \(X, Y\) be independent, continuous random variables. For the sum, we have
    \[
        f_{X + Y}\para{z} = \int_{\minusinfty}^{\infty} f_X\para{x} f_Y\para{z - x} \Diff x.
    \]
\end{proposition}

\begin{proof}
    Investigating the distribution function, we have
    \begin{align*}
        \Prob\para{X + Y \leq z} &= \Prob\para{\para{X, Y} \in \set{\para{x, y} \in \RR^2}{x + y \leq z}} \\
        &= \iint_{x + y \leq z} \Diff^2 \para{x, y} f_{X, Y} \para{x, y} \\
        &= \iint_{x + y \leq z} \Diff^2 \para{x, y} f_{X} \para{x} f_{Y} \para{y} \\
        &= \int_{\minusinfty}^{\plusinfty} \Diff x \int_{\minusinfty}^{z - x} \Diff y f_X\para{x} f_Y\para{y} \\
        &= \int_{\minusinfty}^{\plusinfty} \Diff x \int_{\minusinfty}^{z} \Diff y f_X\para{x} f_Y\para{y - z} \\
        &= \int_{\minusinfty}^{z} \Diff y \para{\int_{\minusinfty}^{\plusinfty} f_X\para{x} f_Y\para{y - x} \Diff x}.
    \end{align*}

    Differentiating this gives the result, as desired.
\end{proof}

\begin{definition}[Convolution of Continuous Random Variables]
    \label{def:convolution-crv}%
    Let \(X, Y\) be two continuous random variables. Then the distribution of \(X + Y\), given by
    \[
        \int_{\minusinfty}^{\infty} f_X\para{x} f_Y\para{z - x} \Diff x
    \]
    is known as the \emph{convolution} of the two random variables.
\end{definition}

\subsection{Conditional Density}

\begin{definition}[Conditional Density]
    \label{def:conditional-density}%
    Let \(X, Y\) be two continuous random variables, with joint density \(f_{X, Y}\) and marginal densities \(f_{X}\) and \(f_{Y}\). The \emph{conditional density} of \(X\) given \(Y = y\) is
    \[
        f_{\bound{\Conditional{X}{Y}}} \para{\Conditional{x}{y}} = \frac{f_{X, Y}\para{x, y}}{f_Y\para{y}}
    \]
    for \(f_Y\para{y} > 0\).
\end{definition}

\begin{theorem}[Continuous Version of Law of Total Probability]
    \label{thm:cont-law-total-prob}%
    For continuous random variables \(X\) and \(Y\), we have
    \[
        f_X\para{x} = \int_{\minusinfty}^{\plusinfty} f_{X, Y} \para{x, y} \Diff y = \int_{\minusinfty}^{\plusinfty} f_{\bound{\Conditional{X}{Y}}} \para{\Conditional{X}{Y}} f_Y\para{y} \Diff y.
    \]
\end{theorem}

\begin{definition}[Conditional Expectation for Continuous Random Variable]
    \label{def:conditional-expectation-crv}%
    For continuous random variables \(X\) and \(Y\), the \emph{conditional expectation} of \(X\) given \(Y\) is defined as
    \[
        \Expt\brac{\Conditional{X}{Y}} = g\para{Y},
    \]
    where we have
    \[
        g\para{y} = \Expt\brac{\Conditional{X}{Y = y}} = \int_{\minusinfty}^{\plusinfty} x f_{\bound{\Conditional{X}{Y}}} \para{\Conditional{x}{y}} \Diff x.
    \]
\end{definition}

\subsection{Transformation of Continuous Random Variables}

\begin{theorem}[Transformation of Joint Continuous Random Variables]
    \label{thm:transformation-joint-continuous-random-variable}%
    Let \(\vm{X}\) be a random variable with values in \(\mathcal{D} \subseteq \RR^n\) and density \(f\). Let
    \[
        \functiondomain{g}{\mathcal{D}}{g\para{\mathcal{D}} \subseteq \RR^n}
    \]
    be a bijection, with a continuous derivative on \(\mathcal{D}\), with
    \[
        \det g'\para{\vm{x}} \neq 0
    \]
    for any \(\vm{x} \in \mathcal{D}\). Then for any \(\vm{y} \in g\para{\mathcal{D}}\), the random variable \(\vm{Y} = g\para{\vm{X}}\) has density
    \[
        f_{\vm{Y}}\para{\vm{y}} = f_{\vm{X}} \para{g\inverse\para{\vm{y}}} \cdot \abs{\det \vm{J}},
    \]
    where \(\vm{J}\) is the \emph{Jacobian}, given by
    \[
        J_{i, j} = \PartialFrac{x_i}{y_j}
    \]
    where \(\vm{y} = g\para{\vm{x}}\).
\end{theorem}

\begin{example}[Transformation of Cartesian and Polar Form]
    \label{ex:transformation-cartesian-polar}%
    Let \(X, Y \sim \Normal\para{0, 1}\) be independent, and represent Cartesian coordinates in the plane. Let \(R, \Theta\) be the random variables for the usual polar coordinates, i.e.,
    \[
        X = R \cos \Theta, \quad Y = R \sin \Theta,
    \]
    restricting us to \(\Theta \in \cointv{0}{2\pi}\).

    Applying \autoref{thm:transformation-joint-continuous-random-variable}, we have7
    \[
        f_{\para{R, \Theta}} \para{r, \theta} = f_{\para{X, Y}} \para{r \cos \theta, r \sin \theta} \cdot \abs{\det \vm{J}},
    \]
    where the Jacobian is given by
    \[
        \vm{J} = \begin{pmatrix}
            \cos \theta & - r \sin \theta \\
            \sin \theta & r \cos \theta
        \end{pmatrix},
    \]
    and so \(\abs{\det \vm{J}} = r\). Hence, using independence,
    \begin{align*}
        f_{\para{R, \Theta}} \para{r, \theta} &= f_X \para{r \cos \theta} \cdot f_Y \para{r \sin \theta} \cdot r \\
        &= \frac{1}{\sqrt{2\pi}} \cdot e^{- \frac{r^2 \cos^2 \theta}{2}} \cdot \frac{1}{\sqrt{2 \pi}} \cdot e^{- \frac{r^2 \sin^2 \theta}{2}} \\
        &= \frac{1}{2\pi} \cdot r \cdot e^{- \frac{r^2}{2}}.
    \end{align*}

    Therefore, \(\Theta \sim \Uniform\ccintv{0}{2\pi}\), and \(R \sim r e^{-\frac{r^2}{2}}\) for \(r > 0\). They are independent.
\end{example}

\subsection{Order Statistics of a Random Variable}

Let \(X_1, \ldots, X_n\) be independent and identically distributed random variables, with probability distribution function \(F\) and density function \(f\).

We order them from smallest to largest, which then gives
\[
    X_{\para{1}} \leq X_{\para{2}} \leq \cdots \leq X_{\para{n}},
\]
and we let \(Y_i = X_{\para{i}}\).

\begin{definition}[Order Statistics]
    \label{def:order-statistics}%
    The \(Y_i\)s found above are the \emph{order statistics} of the sample.
\end{definition}

\begin{example}[Density of \(Y_1\) and \(Y_n\)]
    \label{ex:density-y1-yn}%
    For \(Y_1\), we have
    \[
        \Prob\para{Y_1 \leq x} = 1 - \Prob\para{Y_1 > x} = 1 - \para{\Prob\para{X_1 > x}}^n = 1 - \para{1 - F\para{x}}^n,
    \]
    and hence
    \[
        f_{Y_1} \para{x} = \DiffOp{x} \Prob\para{Y_1 \leq x} = n \para{1 - F\para{x}}^{n - 1} f\para{x}.
    \]

    From a similar argument, we have
    \[
        \Prob\para{Y_n \leq x} = \Prob\para{X_1 \leq x}^n = \para{F\para{x}}^n.
    \]

    Hence, differentiating gives
    \[
        f_{Y_n} \para{x} = n \para{F\para{x}}^{n - 1} f\para{x}.
\]
\end{example}

\begin{proposition}[Density of Order Statistics]
    \label{prop:density-order-stats}%
    We have
    \[
        f_{\para{Y_1, \ldots, Y_n}} \para{x_1, \ldots, x_n} = \begin{cases}
            n! f\para{x_1} \cdots f\para{x_n}, & x_1 < \cdots < x_n, \\
            0, & \text{otherwise}.
        \end{cases}
    \]
\end{proposition}

\begin{proof}
    For \(x_1 < x_2 < \cdots < x_n\), we have
    \begin{align*}
        F_{\para{Y_1, \ldots, Y_n}} \para{x_1, \ldots, x_n} &= \Prob\para{Y_1 \leq x_1, \ldots, Y_n \leq x_n} \\
        &= n! \cdot \Prob\para{X_1 \leq x_1, \ldots, X_n \leq x_n, X_1 \leq \cdots \leq X_n} \\
        &= n! \cdot \int_{\RR} \Diff u_1 \cdots \int_{\RR} \Diff u_n \cdot f_{\para{X_1, \ldots, X_n}} \para{u_1, \ldots, u_n} \cdot \\
        &\phantom{=} \quad \indicator \para{u_1 \leq x_1, \ldots, u_n \leq x_n, u_1 \leq \cdots \leq u_n} \\
        &= n! \cdot \int_{\minusinfty}^{x_1} \Diff u_1 \int_{u_1}^{x_2} \Diff u_2 \cdots \int_{u_{n - 1}}^{x_n} \Diff u_n \cdot f\para{u_1} \cdots f\para{u_n}.
    \end{align*}

    Differentiating gives us
    \[
        f_{\para{Y_1, \ldots, Y_n}} \para{x_1, \ldots, x_n} = \begin{cases}
            n! \cdot f\para{x_1} \cdots f\para{x_n}, & x_1 < \cdots < x_n, \\
            0, & \text{otherwise}.
        \end{cases}
    \]
\end{proof}

In particular, note from the above that the \(Y_i\)s are not independent -- although this density function seems to be separable, it is piecewise where the pieces depend on the relation between the variables.

\begin{example}[Order Statistics for Exponentials]
    \label{ex:order-statistics-exp}%
    Let \(X \sim \Exponential \para{\lambda}\) and \(Y \sim \Exponential \para{\mu}\), with \(X \indp Y\) and \(\lambda, \mu > 0\). Set \(Z = \min\brce{X, Y}\).

    We have for \(z > 0\), that
    \[
        \Prob\para{Z \leq z} = 1 - \Prob\para{Z \geq z} = 1 - \Prob\para{X \geq z} \cdot \Prob\para{Y \geq z} = 1 - e^{- \para{\lambda + \mu} z}.
    \]

    This gives \(Z \sim \Exponential\para{\lambda + \mu}\).

    Similarly, if \(X_i \sim \Exponential \para{\lambda_i}\) are independent, we have
    \[
        \min\brce{X_1, \ldots, X_n} \sim \Exponential\para{\sum_{i = 1}^{n} \lambda_i}.
    \]

    Now, suppose \(\lambda_1 = \cdots = \lambda_n = \lambda\), and let \(Y_1, Y_2, \ldots, Y_n\) be the order statistics.

    We \(Z_1 = Y_1, Z_2 = Y_2 - Y_1, \ldots, Z_i = Y_i - Y_{i - 1}\). We find the density of \(\para{Z_1, \ldots, Z_n}\).

    Let
    \[
        \vm{Z} = \begin{pmatrix}
            Z_1 \\ \vdots \\ Z_n
        \end{pmatrix},
        \vm{Y} = \begin{pmatrix}
            Y_1 \\ \vdots \\ Y_n
        \end{pmatrix},
        \vm{A} = \begin{pmatrix}
             1 &  0 & 0 & \cdots &  0 &  0 &  0 \\
            -1 &  1 & 0 & \cdots &  0 &  0 &  0 \\
             0 & -1 & 1 & \cdots &  0 &  0 &  0 \\
             \vdots & \vdots & \vdots & \ddots & \vdots & \vdots & \vdots \\

             0 &  0 & 0 & \cdots & -1 &  1 & 0 \\
             0 &  0 & 0 & \cdots &  0 & -1 & 1
        \end{pmatrix},
    \]
    so we have \(\vm{Z} = \vm{A} \vm{y}\), and \(\det \vm{A} = 1\). Hence, let \(\vm{z} = \vm{A} \vm{y}\), we have
    \begin{align*}
        f_{\para{Z_1, \ldots, Z_n}} \para{z_1, \ldots, z_n} &= f_{\para{Y_1, \ldots, Y_n}} \para{y_1, \ldots, y_n} \abs{\det \vm{A}} \\
        &= n! \cdot f\para{y_1} \cdots f\para{y_n} \\
        &= n! \lambda e^{-\lambda y_1} \cdots \lambda e^{-\lambda y_n} \\
        &= \prod_{i = 1}^{n} \para{\para{n - i + 1} \lambda e^{-\lambda \para{n - i + 1} z_i}}.
    \end{align*}

    So the \(Z_i\)s are independent, and
    \[
        Z_i \sim \Exponential\para{\lambda \para{n - i + 1}}.
    \]
\end{example}

\subsection{Expectation and Variance Algebra}

\begin{definition}[Expectation and Variance of Multivariate Random Variable]
    \label{def:expt-var-multivariate}%
    For a random variable \(\vm{X}\), we define the \emph{mean}
    \[
        \vmb{\mu} = \Expt\brac{\vm{X}} = \begin{pmatrix}
            \Expt\brac{X_1} \\ \vdots \\ \Expt\brac{X_n}
        \end{pmatrix},
    \]
    and the \emph{variance}
    \[
        \Var\para{\vm{X}} = \Expt\brac{\para{\vm{X} - \vmb{\mu}} \para{\vm{X} - \vmb{\mu}}\transpose}.
    \]

    Equivalently, the variance is defined as
    \[
        \Var\para{\vm{X}} = \para{\Cov\para{X_i, X_j}}_{i, j}.
    \]
\end{definition}

\begin{proposition}[Expectation and Variance of Multivariate Random Variable]
    \label{prop:expt-var-transform-multivariate-rv}%
    We have
    \[
        \Expt\brac{\vm{u} \transpose \vm{X}} = \vm{u}\transpose \Expt\brac{\vm{X}}
    \]
    and
    \[
        \Var\para{\vm{u} \transpose \vm{X}} = \vm{u}\transpose \Var\para{\vm{X}} \vm{u}.
    \]
\end{proposition}

\begin{proof}
    \[
        \Expt\brac{\vm{u} \transpose \vm{X}} = \Expt\brac{\sum_{i = 1}^{n} u_i X_i} = \sum_{i = 1}^{n} u_i \Expt\brac{X_i} = \vm{u}\transpose \Expt\brac{X},
    \]
    and
    \[
        \Var\para{\vm{u} \transpose \vm{X}} = \Var\para{\sum_{i = 1}^{n} u_i X_i} = \sum_{i, j = 1}^{n} u_i \Cov\para{X_i, X_j} u_j = \vm{u}\transpose \vm{\Var\para{\vm{X}}} \vm{u}. 
    \]
\end{proof}

\begin{definition}[Positive/Negative Definiteness]
    \label{def:positive-negative-definiteness}%
    Let \(\vm{M}\) be an \(n \times n\) real symmetric matrix. Then we say it is
    \begin{itemize}
        \item \emph{positive definite}, denoted as \(\vm{M} \posDef 0\), if for all \(\vm{u} \in \RR^n \setminus \brce{\vm{0}}\), we have \(\vm{u}\transpose \vm{M} \vm{u} > 0\);
        \item \emph{positive semi-definite}, or \emph{non-negative definite}, denoted as \(\vm{M} \posDefEq 0\), if for all \(\vm{u} \in \RR^n\), we have \(\vm{u} \transpose \vm{M} \vm{u} \geq 0\);
        \item \emph{negative definite}, denoted as \(\vm{M} \negDef 0\), if for all \(\vm{u} \in \RR^n \setminus \brce{\vm{0}}\), we have \(\vm{u}\transpose \vm{M} \vm{u} < 0\);
        \item \emph{negative semi-definite}, or \emph{non-positive definite}, denoted as \(\vm{M} \negDefEq 0\), if for all \(\vm{u} \in \RR^n\), we have \(\vm{u} \transpose \vm{M} \vm{u} \leq 0\).
    \end{itemize}

    Equivalently, in terms of eigenvalues, it is
    \begin{itemize}
        \item \emph{positive definite}, if all its eigenvalues are positive;
        \item \emph{positive semi-definite}, or \emph{non-negative definite}, if all its eigenvalues are non-negative;
        \item \emph{negative definite}, if all its eigenvalues are negative;
        \item \emph{negative semi-definite}, or \emph{non-positive definite}, if all its eigenvalues are non-positive.
    \end{itemize}
\end{definition}

\begin{claim}[Variance is Non-Negative Definite]
    \label{claim:variance-non-negative-definite}%
    The variance of a multivariate distribution is non-negative definite.
\end{claim}

\begin{proof}
    We have
    \[
        \Var\para{\vm{u}\transpose \vm{X}} = \vm{u}\transpose \Var\para{\vm{X}} \vm{u} \geq 0.
    \]
\end{proof}

\begin{definition}[Square Root of Non-Negative Definite Matrix]
    \label{def:sqrt-non-negative-definite-matrix}%
    Let \(\vm{V}\) be a non-negative definite matrix, and let its orthogonal decomposition be
    \[
        \vm{V} = \vm{U}\transpose \vm{D} \vm{U}
    \]
    for an orthogonal matrix \(\vm{U}\transpose = \vm{U}\inverse\), and \(\vm{D}\) diagonal with
    \[
        \vm{D} = \diag\para{\lambda_1, \ldots, \lambda_n} = \begin{pmatrix}
            \lambda_1 & & 0 \\
             & \ddots & \\
            0 & & \lambda_n
        \end{pmatrix}
    \]
    with \(\lambda_i \geq 0\) for \(i = 1, \ldots, n\).

    The \emph{square root} of \(\vm{V}\) is defined as
    \[
        \vmb{\sigma} = \vm{U}\transpose \sqrt{\vm{D}} \vm{U}
    \]
    where
    \[
        \sqrt{\vm{D}} = \diag\para{\sqrt{\lambda_1}, \ldots, \sqrt{\lambda_n}} = \begin{pmatrix}
            \sqrt{\lambda_1} & & 0 \\
             & \ddots & \\
            0 & & \sqrt{\lambda_n}
        \end{pmatrix}
    \]
\end{definition}
